# Gatecast: every public page may be crawled, including the plain book pages under /stacks/. The live-data API is
# not a set of pages, but the three endpoints the front page and The Stacks read are allowed so a search engine's
# renderer sees the same page a reader does (their JSON is kept out of the index by an X-Robots-Tag noindex header,
# vercel.json). The reviewer console is not blocked here on purpose: it sends X-Robots-Tag noindex, nofollow, and a
# crawler blocked by robots.txt would never see that header.
User-agent: *
Allow: /
Allow: /stacks/
Allow: /api/weird
Allow: /api/news
Allow: /api/book-covers
Disallow: /api/
